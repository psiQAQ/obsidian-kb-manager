<%*
const root = "{{kb_root}}";
const file = tp.config.target_file;
if (!file?.path.startsWith(root + "/Notes/") || file.parent?.path !== root + "/Notes") throw new Error("请在本库 Notes 直接打开待归档札记。");
const properties = app.metadataCache.getFileCache(file)?.frontmatter;
if (properties?.type !== "jot" || !/^\d{4}-\d{2}-\d{2}$/.test(properties.date)) throw new Error("请先为札记填入 YYYY-MM-DD 日期。");
const folder = root + "/Notes/" + properties.date.slice(0, 4);
if (!app.vault.getAbstractFileByPath(folder)) await app.vault.createFolder(folder);
const destination = folder + "/" + file.name;
if (app.vault.getAbstractFileByPath(destination)) throw new Error("同一天所在年度已有同名笔记。");
await app.fileManager.renameFile(file, destination);
