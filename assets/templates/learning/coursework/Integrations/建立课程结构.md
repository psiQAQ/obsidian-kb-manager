<%*
const root = "{{kb_root}}";
if (tp.config.target_file?.path !== root + "/README.md") throw new Error("请从本库课程首页启动。");
const name = (await tp.system.prompt("课程名"))?.trim();
if (!name) return;
if (/^[.]|[.]$|[<>:"/\\|?*\u0000-\u001f]|^(con|prn|aux|nul|com[1-9¹²³]|lpt[1-9¹²³])(?:[.]|$)/iu.test(name)) throw new Error("课程名含不支持的文件名字符。");
const directory = root + "/Courses/" + name;
for (const folder of [directory, ...["Lectures", "Assignments", "Notes"].map(role => directory + "/" + role)]) {
	const existing = app.vault.getAbstractFileByPath(folder);
	if (existing && !Array.isArray(existing.children)) throw new Error("课程目录被同名文件占用：" + folder);
	if (!existing) await app.vault.createFolder(folder);
}
const entry = directory + "/课程索引.md";
const existing = app.vault.getAbstractFileByPath(entry);
if (existing && existing.extension !== "md") throw new Error("课程索引被同名目录占用。");
const course = existing ?? await app.vault.create(entry, "---\ntype: course\nstatus: active\n---\n\n# " + name + "\n\n## 讲义\n\n## 作业\n");
await app.workspace.getLeaf('tab').openFile(course);
%>
